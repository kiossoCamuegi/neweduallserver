 async function getOperatorData(res, userId) {
   if (!userId) return null;
 
   try {
     const query = `
       SELECT ed_user_account_id, ed_user_account_name
       FROM eduall_user_accounts
       WHERE ed_user_account_id = ?
         AND ed_user_account_deleted = 0
       LIMIT 1
     `;
 
     const [operator] = await ExecuteQuery(res, query, [userId], true, {
       db: "eduall_user_accounts",
       type: "select",
       data: { operatorLookup: true }
     });
 
     if (operator) {
       return {
         id: operator.ed_user_account_id,
         name: operator.ed_user_account_name || "Operador Desconhecido"
       };
     }
 
     return null;
   } catch (error) {
     console.warn("Erro ao buscar dados do operador:", error.message);
     return null;
   }
 }

 const GetFeepayments = async (req, res) => {
   try {
     const institute = req.session.user.eduall_user_session_curentinstitute;
     const operatorId = req.session?.user?.eduall_user_session_ID;
     const operator = await getOperatorData(res, operatorId);
 
     const query = `
       SELECT 
         efp.*,
         st.ed_student_id,
         st.ed_student_name,
         st.ed_student_code,
         s.ed_service_id,
         s.ed_service_title,
         c.ed_coin_code,
         c.ed_coin_code,
         ua.ed_user_account_name AS operator_name,
         ua.ed_user_account_id AS operator_id,
         (
           SELECT JSON_ARRAYAGG(
             JSON_OBJECT(
               'id', ex.ed_fee_additional_id,
               'description', ex.description,
               'amount', ex.amount,
               'created_at', ex.created_at
             )
           )
           FROM eduall_fee_additional_items ex
           WHERE ex.invoice_code = efp.ed_fee_payment_invoiceCode
             AND ex.institute_code = efp.ed_fee_payment_institute_code
             AND COALESCE(ex.ed_fee_additional_deleted,0)=0
         ) AS extras
       FROM eduall_fees_payments efp
       LEFT JOIN eduall_students st
         ON efp.ed_fee_payment_student = st.ed_student_id
         AND st.ed_student_deleted = 0
       LEFT JOIN eduall_services s
         ON efp.ed_fee_payment_service = s.ed_service_id
         AND s.ed_service_deleted = 0
       LEFT JOIN eduall_coins c
         ON c.ed_coin_institute_code = ?
       LEFT JOIN eduall_user_accounts ua
         ON ua.ed_user_account_id = ?
         AND ua.ed_user_account_deleted = 0
       WHERE efp.ed_fee_payment_deleted = 0
         AND efp.ed_fee_payment_institute_code = ?
       ORDER BY efp.ed_fee_payment_registerDate DESC
     `;
 
     const PARAMS = [institute, operatorId, institute];
 
     const rows = await ExecuteQuery(res, query, PARAMS, true, {
       db: "eduall_fees_payments",
       type: "select",
       data: req
     });
 
     // Adicionar informação de operador
     const enrichedRows = rows.map(row => ({
       ...row,
       operator: operator?.name || "operator:null",
       operator_id: operator?.id || null
     }));
 
     res.status(200).json(enrichedRows);
   } catch (error) {
     res.status(500).json({
       success: false,
       message: "Falha ao buscar os pagamentos de taxa!",
       error: error.message
     });
   }
 };
 
 

const RegisterFeepayment = async (req, res) => {
 
  // ── Helpers internos ────────────────────────────────────────────────────────
  
  /**
   * Calcula totais de uma linha (desconto, multa, IVA)
   */
  function computeItemTotals({ 
    price, 
    discountPct = 0, 
    fineType = 0, 
    fineValue = 0, 
    applyIVA = false, 
    instituteIVAPct = 0 
  }) {
    const discountAmount     = discountPct > 0 ? (price * discountPct) / 100 : 0;
    const priceAfterDiscount = Math.max(0, price - discountAmount);
    const fineAmount         = fineType * 1 === 1
      ? fineValue * 1
      : fineValue > 0 ? (priceAfterDiscount * fineValue) / 100 : 0;
    const ivaAmount          = applyIVA ? (priceAfterDiscount * instituteIVAPct) / 100 : 0;
    const lineTotal          = priceAfterDiscount + fineAmount + ivaAmount;
    
    return { 
      discountAmount, 
      fineAmount, 
      ivaAmount, 
      lineTotal, 
      baseAfterDiscount: priceAfterDiscount 
    };
  }
 
  /**
   * Formata mês (formato: "0.2025" → "Janeiro / 2025")
   */
  function monthLabel(mdotY) {
    const m = String(mdotY).split(".")[0] * 1;
    const y = String(mdotY).split(".")[1] * 1;
    const N = [
      "Janeiro", "Fevereiro", "Março", "Abril", "Maio", "Junho",
      "Julho", "Agosto", "Setembro", "Outubro", "Novembro", "Dezembro"
    ];
    if (Number.isFinite(m) && Number.isFinite(y)) return `${N[m]} / ${y}`;
    return String(mdotY);
  }
 
  /**
   * Valida e sanitiza serviços extras
   */
  function sanitizeExtras(extras) {
    if (!Array.isArray(extras)) return [];
    return extras
      .map((e) => ({
        name:  String(e?.name  || "").trim(),
        value: Number(String(e?.value ?? "").toString().replace(",", ".")),
      }))
      .filter((e) => e.name.length > 0 && Number.isFinite(e.value) && e.value > 0);
  }
 
  // ── Corpo principal ─────────────────────────────────────────────────────────
  try {
    const sessionInstitute = req.session?.user?.eduall_user_session_curentinstitute;
    const operatorId       = req.session?.user?.eduall_user_session_ID;
    const operator         = await getOperatorData(res, operatorId);
 
    if (!sessionInstitute) {
      return res.status(401).json({ success: false, msg: "Sessão inválida." });
    }
 
    const {
      studentId, 
      academicYearId, 
      paymentType,
      bankName     = "",
      paymentPlace = "",
      bordereux    = "",
      items        = [],
      extraServices = [],
      balanceUsed  = 0,  // ← NOVO: créditos aplicados pelo utilizador (pode ser PARCIAL)
    } = req.body || {};
 
    if (!studentId || !academicYearId || !Array.isArray(items) || !items.length) {
      return res.status(400).json({ success: false, msg: "Dados incompletos." });
    }
 
    // Sanitizar e normalizar entrada
    const addons      = sanitizeExtras(extraServices);
    const balanceUsedValue = Math.max(0, parseFloat(balanceUsed) || 0);
 
    // ─────────────────────────────────────────────────────────────────────────
    // 1) BUSCAR DADOS DO ALUNO
    // ─────────────────────────────────────────────────────────────────────────
    const studentSql = `
      SELECT
        s.ed_student_id, s.ed_student_code, s.ed_student_name, s.ed_student_identityCard,
        s.ed_student_address, s.ed_student_email, s.ed_student_class, s.ed_student_institute_code,
        cls.ed_class_title, room.ed_classroom_title, crs.ed_course_title, ay.ed_academic_year_title
      FROM eduall_students s
      LEFT JOIN eduall_class cls ON cls.ed_class_id = s.ed_student_class AND COALESCE(cls.ed_class_deleted,0)=0
      LEFT JOIN eduall_classrooms room ON room.ed_classroom_id = cls.ed_class_room AND COALESCE(room.ed_classroom_deleted,0)=0
      LEFT JOIN eduall_courses crs ON crs.ed_course_id = cls.ed_class_course AND COALESCE(crs.ed_course_deleted,0)=0
      LEFT JOIN eduall_academic_year ay ON ay.ed_academic_year_id = cls.ed_class_year AND COALESCE(ay.ed_academic_year_deleted,0)=0
      WHERE COALESCE(s.ed_student_deleted,0)=0
        AND s.ed_student_id = ?
        AND s.ed_student_institute_code = ?
      LIMIT 1
    `;
    const [student] = await ExecuteQuery(res, studentSql, [studentId, sessionInstitute], true);
    if (!student) {
      return res.status(404).json({ success: false, msg: "Estudante não encontrado." });
    }
 
    // ─────────────────────────────────────────────────────────────────────────
    // 2) VERIFICAR DUPLICATAS DE MESES
    // ─────────────────────────────────────────────────────────────────────────
    const months        = [...new Set(items.map((i) => i.month))];
    const placeholdersM = months.map((_) => "?").join(",");
    const dupSql        = `
      SELECT ed_fee_payment_month AS month
      FROM eduall_fees_payments
      WHERE ed_fee_payment_institute_code = ?
        AND ed_fee_payment_student        = ?
        AND ed_fee_payment_academic_year  = ?
        AND ed_fee_payment_month IN (${placeholdersM})
        AND COALESCE(ed_fee_payment_deleted,0) = 0
      GROUP BY ed_fee_payment_month
    `;
    const dupRows = await ExecuteQuery(
      res, 
      dupSql, 
      [sessionInstitute, studentId, academicYearId, ...months], 
      true
    );
    if (dupRows?.length) {
      const paidLabels = dupRows.map((r) => monthLabel(r.month));
      return res.status(409).json({
        success: false,
        code: "DUPLICATE_MONTHS",
        msg: `Os seguintes meses já foram pagos: ${paidLabels.join(", ")}.`,
        monthsPaid: dupRows.map((r) => String(r.month)),
      });
    }
 
    // ─────────────────────────────────────────────────────────────────────────
    // 3) BUSCAR IVA DA INSTITUIÇÃO
    // ─────────────────────────────────────────────────────────────────────────
    const ivaSql      = `SELECT COALESCE(ed_institute_iva_value,0) AS iva, ed_institute_name FROM eduall_institutes WHERE ed_institute_code=? LIMIT 1`;
    const [ivaRow]    = await ExecuteQuery(res, ivaSql, [sessionInstitute], true);
    const instituteIVAPct = ivaRow?.iva * 1 || 0;
    const instituteName   = ivaRow?.ed_institute_name || "";
 
    // ─────────────────────────────────────────────────────────────────────────
    // 4) CALCULAR TOTAIS POR LINHA E SOMAR TOTAL GERAL
    // ─────────────────────────────────────────────────────────────────────────
    const invoiceLines = [];
    const dbRows       = [];
    let grandTotal     = 0;
 
    for (const it of items) {
      const { serviceId, month, price, discountPct = 0, fineType = 0, fineValue = 0, applyIVA = 0 } = it;
      const calc = computeItemTotals({
        price: price * 1, 
        discountPct, 
        fineType, 
        fineValue,
        applyIVA: !!applyIVA, 
        instituteIVAPct,
      });
 
      // Linha da propina
      invoiceLines.push({
        itemcode: `EDSERV-${serviceId}`,
        title:    `Propina ${monthLabel(month)}`,
        price:    Number(calc.baseAfterDiscount.toFixed(2)),
        tax:      applyIVA ? 1 : 0,
        discount: 0,
        quantity: "1",
      });
 
      // Linha da multa (se houver)
      if (calc.fineAmount > 0) {
        invoiceLines.push({
          itemcode: "FINE001",
          title:    "Multa por atraso",
          price:    Number(calc.fineAmount.toFixed(2)),
          tax:      0, 
          discount: 0, 
          quantity: "1",
        });
      }
 
      grandTotal += calc.lineTotal;
 
      // Registar para inserção na BD
      dbRows.push({
        studentId, 
        serviceId,
        price:       price * 1,
        month,
        discountPct,
        type:        paymentType,
        bankName,
        academicYearId,
        place:       paymentPlace,
        balance:     Number(calc.lineTotal.toFixed(2)),
        bordereux,
        ivaPct:      applyIVA ? instituteIVAPct : 0,
        fineType,
        fineValue:   Number(calc.fineAmount.toFixed(2)),
      });
    }
 
    // Adicionar serviços extras ao total
    if (addons.length) {
      addons.forEach((ex, idx) => {
        invoiceLines.push({
          itemcode: `EXTRA-${String(idx + 1).padStart(2, "0")}`,
          title:    ex.name,
          price:    Number(Number(ex.value).toFixed(2)),
          tax:      0, 
          discount: 0, 
          quantity: "1",
        });
        grandTotal += ex.value * 1;
      });
    }
 
    // ─────────────────────────────────────────────────────────────────────────
    // 5) CONSTRUIR OBSERVAÇÕES (inclui detalhes do saldo PARCIAL se aplicável)
    // ─────────────────────────────────────────────────────────────────────────
    const obsParts = [];
    obsParts.push(`Nº matrícula: ${student.ed_student_code || "-"}`);
    if (student.ed_class_title)        obsParts.push(`Turma: ${student.ed_class_title}`);
    if (student.ed_classroom_title)    obsParts.push(`Sala: ${student.ed_classroom_title}`);
    if (student.ed_course_title)       obsParts.push(`Curso: ${student.ed_course_title}`);
    if (student.ed_academic_year_title) obsParts.push(`Ano letivo: ${student.ed_academic_year_title}`);
    if (instituteName)                 obsParts.push(`Instituição: ${instituteName}`);
    if (operator)                      obsParts.push(`Operador: ${operator.name}`);
 
    // ── BLOCO CRÍTICO: DETALHAR O SALDO PARCIAL ──────────────────────────────
    if (Number(paymentType) === PAYMENT_TYPE_BALANCE && balanceUsedValue > 0) {
      const coin            = ""; // A moeda vem do campo da fatura; aqui apenas numérico
      const totalFormatted  = grandTotal.toFixed(2);
      const balanceFormatted = balanceUsedValue.toFixed(2);
      const remaining       = Math.max(0, grandTotal - balanceUsedValue);
 
      obsParts.push("---");
      obsParts.push("Forma de pagamento: Saldo do aluno (PARCIAL)");
      obsParts.push(`Total da fatura: ${totalFormatted}`);
      obsParts.push(`Saldo do aluno utilizado: ${balanceFormatted}`);
 
      if (remaining > 0) {
        obsParts.push(`Valor restante (a cobrar por outro método): ${remaining.toFixed(2)}`);
      } else {
        obsParts.push("Fatura totalmente coberta pelo saldo do aluno.");
      }
    }
 
    const obs = obsParts.join("\n");
 
    // ─────────────────────────────────────────────────────────────────────────
    // 6) DADOS DO CLIENTE PARA A FATURA
    // ─────────────────────────────────────────────────────────────────────────
    const customer = {
      name:      student.ed_student_name         || "",
      fiscal_id: student.ed_student_identityCard || "",
      address:   student.ed_student_address      || "",
      email:     student.ed_student_email        || "",
    };
 
    // ─────────────────────────────────────────────────────────────────────────
    // 7) CRIAR FATURA (Vendus / sistema de faturação externo)
    // ─────────────────────────────────────────────────────────────────────────
    const vendus = await VENDUS_CreateInvoiceCodeAndDoc(
      req, 
      customer, 
      invoiceLines, 
      paymentType, 
      obs
    );
    if (!vendus?.number) {
      return res.status(500).json({ success: false, msg: "Falha ao criar fatura." });
    }
 
    // ─────────────────────────────────────────────────────────────────────────
    // 8) TRANSAÇÃO NA BASE DE DADOS
    // ─────────────────────────────────────────────────────────────────────────
    await ExecuteQuery(res, "START TRANSACTION", [], false);
 
    try {
      // 8.a) INSERIR LINHAS DE PROPINA
      const insertSql = `
        INSERT INTO eduall_fees_payments (
          ed_fee_payment_student,
          ed_fee_payment_service,
          ed_fee_payment_price,
          ed_fee_payment_month,
          ed_fee_payment_discount,
          ed_fee_payment_type,
          ed_fee_payment_bank,
          ed_fee_payment_academic_year,
          ed_fee_payment_place,
          ed_fee_payment_balance,
          ed_fee_payment_bordereux,
          ed_fee_payment_iva,
          ed_fee_payment_code,
          ed_fee_payment_fineType,
          ed_fee_payment_fineValue,
          ed_fee_payment_invoiceCode,
          ed_fee_payment_invoiceFile,
          ed_fee_payment_institute_code,
          ed_fee_payment_operator,
          ed_fee_payment_operator_id
        ) VALUES
        ${dbRows.map(() => "(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)").join(",")}
      `;
      const params = [];
      for (const r of dbRows) {
        params.push(
          r.studentId, 
          r.serviceId, 
          r.price, 
          r.month, 
          r.discountPct, 
          r.type, 
          r.bankName,
          r.academicYearId, 
          r.place, 
          r.balance, 
          r.bordereux, 
          r.ivaPct, 
          null, 
          r.fineType,
          r.fineValue, 
          vendus.number, 
          vendus.id, 
          sessionInstitute,
          operator?.name || "operator:null",
          operator?.id   || null
        );
      }
      await ExecuteQuery(res, insertSql, params, false);
 
      // 8.b) PERSISTIR EXTRAS
      if (addons.length) {
        try {
          const addSql = `
            INSERT INTO eduall_fee_additional_items
              (student_id, academic_year_id, description, amount, invoice_code, invoice_id, institute_code, operator, operator_id)
            VALUES
              ${addons.map(() => "(?,?,?,?,?,?,?,?,?)").join(",")}
          `;
          const addParams = [];
          addons.forEach((a) => {
            addParams.push(
              studentId, 
              academicYearId, 
              a.name,
              Number(Number(a.value).toFixed(2)),
              vendus.number, 
              vendus.id, 
              sessionInstitute,
              operator?.name || "operator:null",
              operator?.id   || null
            );
          });
          await ExecuteQuery(res, addSql, addParams, false);
        } catch (auxErr) {
          console.warn("Persistência de extras ignorada:", auxErr?.message || auxErr);
        }
      }
 
      // ─────────────────────────────────────────────────────────────────────
      // 8.c) DEBITAR SALDO DO ALUNO (SE PAGAMENTO POR SALDO & SALDO > 0)
      // ─────────────────────────────────────────────────────────────────────
      if (Number(paymentType) === PAYMENT_TYPE_BALANCE && balanceUsedValue > 0) {
        try {
          await ensureBalanceRow(res, studentId, sessionInstitute);
 
          // LOCK + LER SALDO ATUAL
          const [current] = await ExecuteQuery(
            res,
            `SELECT sb_balance, sb_currency FROM eduall_student_balance
             WHERE sb_student_id = ? AND sb_institute_code = ? FOR UPDATE`,
            [studentId, sessionInstitute],
            true
          );
 
          const balanceBefore = parseFloat(current?.sb_balance ?? 0);
          const currency      = current?.sb_currency || "AOA";
 
          // ✓ VALIDAÇÃO CRÍTICA: saldo disponível ≥ saldo a usar
          if (balanceUsedValue > balanceBefore) {
            // Rejeitar se operador tentar usar mais do que existe
            await ExecuteQuery(res, "ROLLBACK", [], false);
            return res.status(422).json({
              success: false,
              msg: `Saldo insuficiente. Saldo atual: ${balanceBefore} ${currency}. Tentou usar: ${balanceUsedValue} ${currency}.`,
              balance: balanceBefore,
              requested: balanceUsedValue,
            });
          }
 
          const balanceAfter = Number((balanceBefore - balanceUsedValue).toFixed(2));
          const remaining    = Math.max(0, grandTotal - balanceUsedValue);
 
          // ─ DESCRIÇÃO DETALHADA DO LOG ─
          let deductDesc = `Pagamento parcial de propina(s) — Fatura ${vendus.number}. `;
          deductDesc    += `Créditos usados: ${balanceUsedValue.toFixed(2)} ${currency}`;
          deductDesc    += ` (total da fatura: ${grandTotal.toFixed(2)} ${currency})`;
          if (remaining > 0) {
            deductDesc  += `. Restante a pagar por outro método: ${remaining.toFixed(2)} ${currency}.`;
          } else {
            deductDesc  += `. Fatura totalmente coberta pelo saldo.`;
          }
 
          // ATUALIZAR SALDO
          await ExecuteQuery(
            res,
            `UPDATE eduall_student_balance SET sb_balance = ?
             WHERE sb_student_id = ? AND sb_institute_code = ?`,
            [balanceAfter, studentId, sessionInstitute],
            false
          );
 
          // REGISTAR LOG DETALHADO
          await ExecuteQuery(
            res,
            `INSERT INTO eduall_student_balance_logs
               (sb_log_student_id, sb_log_institute, sb_log_type, sb_log_category,
                sb_log_amount, sb_log_balance_before, sb_log_balance_after,
                sb_log_description, sb_log_reference, sb_log_user_id)
             VALUES (?,?,?,?,?,?,?,?,?,?)`,
            [
              studentId, 
              sessionInstitute, 
              "deduct", 
              "propinas",
              balanceUsedValue, 
              balanceBefore, 
              balanceAfter,
              deductDesc, 
              vendus.number || null,
              operatorId || null,
            ],
            false
          );
 
        } catch (balanceErr) {
          // Falha não-crítica no saldo — apenas registar aviso
          console.error(
            "Erro ao debitar saldo (não crítico):", 
            balanceErr?.message || balanceErr
          );
        }
      }
 
      // ─────────────────────────────────────────────────────────────────────
      // 9) AUDITORIA
      // ─────────────────────────────────────────────────────────────────────
      const auditDesc = `[${operator?.name || "Desconhecido"}] Registou ${dbRows.length} linha(s) + ${addons.length} extra(s) para aluno ${student.ed_student_code}. Fatura ${vendus.number}. Total: ${grandTotal.toFixed(2)}${balanceUsedValue > 0 ? `. Saldo usado: ${balanceUsedValue.toFixed(2)}` : ""}.`;
      
      await RegisterAuditAction({
        session:     req.session,
        action:      1,
        form:        "Pagamentos de propinas (lote)",
        description: auditDesc,
      });
 
      // COMMIT DA TRANSAÇÃO
      await ExecuteQuery(res, "COMMIT", [], false);
 
      // ─────────────────────────────────────────────────────────────────────
      // 10) RESPOSTA AO CLIENTE
      // ─────────────────────────────────────────────────────────────────────
      return res.status(200).json({
        success:        true,
        invoice:        vendus,
        inserted:       dbRows.length,
        extrasInserted: addons.length,
        extras:         addons,
        total:          Number(grandTotal.toFixed(2)),
        balanceUsed:    balanceUsedValue > 0 ? Number(balanceUsedValue.toFixed(2)) : 0,
        remaining:      Math.max(0, grandTotal - balanceUsedValue),
        operator:       operator?.name || "operator:null",
      });
 
    } catch (txErr) {
      // Falha na transação — fazer rollback
      await ExecuteQuery(res, "ROLLBACK", [], false);
      throw txErr;
    }
 
  } catch (err) {
    try { await ExecuteQuery(res, "ROLLBACK", [], false); } catch {}
    console.error("RegisterFeepayment error:", err);
    return res.status(500).json({ 
      success: false, 
      msg: "Erro no processamento.", 
      error: String(err?.message || err) 
    });
  }
};
 


const GetSingleFeepayment = async (req, res) => {
  try {
    const { ID } = req.params;
    const institute = req.session.user.eduall_user_session_curentinstitute;

    const query = `
      SELECT 
        p.*, srv.ed_service_title, c.ed_coin_code,
        (
          SELECT JSON_ARRAYAGG(
            JSON_OBJECT(
              'id', ex.ed_fee_additional_id,
              'description', ex.description,
              'amount', ex.amount
            )
          )
          FROM eduall_fee_additional_items ex
          WHERE ex.invoice_code = p.ed_fee_payment_invoiceCode
            AND ex.institute_code = p.ed_fee_payment_institute_code
            AND COALESCE(ex.ed_fee_additional_deleted,0)=0
        ) AS extras
      FROM eduall_fees_payments p
      LEFT JOIN eduall_services srv ON p.ed_fee_payment_service = srv.ed_service_id
      LEFT JOIN eduall_coins c ON c.ed_coin_institute_code = ?
      WHERE p.ed_fee_payment_deleted = 0
        AND p.ed_fee_payment_id = ?
        AND p.ed_fee_payment_institute_code = ?
      LIMIT 1;
    `;

    const PARAMS = [institute, ID, institute];
    const rows = await ExecuteQuery(res, query, PARAMS, true, {
      db: "eduall_fees_payments", type: "select", data: req
    });

    res.status(200).json(rows[0] || {});
  } catch (error) {
    res.status(500).json({
      success: false,
      message: "Falha ao buscar o pagamento de taxa!",
      error: error.message
    });
  }
};


const GetSinglePaidMonth = async (req, res) => {

const FeePaymentDelete = async (req, res) => {
  const { ID } = req.params;
  const institute = req.session?.user?.eduall_user_session_curentinstitute;

  if (!institute) {
    return res.status(400).json({ success: false, message: "Instituto inválido na sessão." });
  }

  try {
    // Inicia transação
    await ExecuteQuery(res, "START TRANSACTION", [], false, { db: "", type: "tx-begin" });

    // 1️⃣ Buscar o código da fatura (invoice_code) antes de marcar como deletado
    const findInvoiceSQL = `
      SELECT ed_fee_payment_invoiceCode AS invoice_code
      FROM eduall_fees_payments
      WHERE ed_fee_payment_id = ? 
        AND ed_fee_payment_institute_code = ? 
        AND COALESCE(ed_fee_payment_deleted,0)=0
      LIMIT 1;
    `;
    const [payment] = await ExecuteQuery(res, findInvoiceSQL, [ID, institute], true, {
      db: "eduall_fees_payments", type: "select", data: req
    });

    if (!payment) {
      await ExecuteQuery(res, "ROLLBACK", [], false, { db: "", type: "tx-rollback" });
      return res.status(404).json({ success: false, message: "Pagamento não encontrado ou já eliminado." });
    }

    const invoiceCode = payment.invoice_code;

    // 2️⃣ Marcar o pagamento como deletado
    const deletePaymentSQL = `
      UPDATE eduall_fees_payments 
      SET ed_fee_payment_deleted = 1
      WHERE ed_fee_payment_id = ? 
        AND ed_fee_payment_institute_code = ? 
        AND COALESCE(ed_fee_payment_deleted,0)=0;
    `;
    await ExecuteQuery(res, deletePaymentSQL, [ID, institute], false, { db: "eduall_fees_payments", type: "update", data: req });

    // 3️⃣ Marcar serviços adicionais ligados à mesma fatura
    if (invoiceCode) {
      const deleteExtrasSQL = `
        UPDATE eduall_fee_additional_items
        SET ed_fee_additional_deleted = 1
        WHERE invoice_code = ?
          AND institute_code = ?
          AND COALESCE(ed_fee_additional_deleted,0)=0;
      `;
      await ExecuteQuery(res, deleteExtrasSQL, [invoiceCode, institute], false, { db: "eduall_fee_additional_items", type: "update", data: req });
    }

    // 4️⃣ Registrar ação de auditoria
    await RegisterAuditAction({
      session: req.session,
      action: 3, // Eliminação
      form: "Pagamentos de propinas",
      description: `Eliminou o pagamento de propina com ID ${ID}${invoiceCode ? ` (fatura ${invoiceCode})` : ""}, incluindo serviços adicionais.`
    });

    // 5️⃣ Commit
    await ExecuteQuery(res, "COMMIT", [], false, { db: "", type: "tx-commit" });

    res.status(200).json({
      success: true,
      message: `Pagamento (ID ${ID}) e serviços adicionais associados eliminados com sucesso.`,
    });

  } catch (error) {
    try { await ExecuteQuery(res, "ROLLBACK", [], false, { db: "", type: "tx-rollback" }); } catch {}
    console.error("❌ Erro ao deletar pagamento:", error);
    res.status(500).json({
      success: false,
      message: "Falha ao deletar o pagamento e seus serviços adicionais!",
      error: error.message || error
    });
  }
};

const GetSingleStudentPaymentStatus = async (req, res) => {
  const timerLabel = `GetSingleStudentPaymentStatus_${Date.now()}`;
  console.time(timerLabel);

  try {
    const institute = req.session?.user?.eduall_user_session_curentinstitute;
    if (!institute) {
      console.warn("⚠️ Instituto não encontrado na sessão."); 
      return res.status(400).json({ success: false, message: "Instituto inválido na sessão." });
    }

    const { ano } = req.query;
    const { studentCode } = req.params;

    if (!studentCode || !ano) { 
      return res.status(400).json({
        success: false,
        message: "Parâmetros inválidos: envie o código do aluno e o ano letivo (?ano=2025)."
      });
    }

    // === Buscar aluno + dados de turma + ano letivo em única query ===
    const studentQuery = `
      SELECT 
        s.ed_student_id, s.ed_student_code, s.ed_student_name, s.ed_student_picture,
        c.ed_class_title, c.ed_class_year, 
        y.ed_academic_year_title, y.ed_academic_year_startDate, y.ed_academic_year_endDate,
        crs.ed_course_title,
        lvl.ed_academic_level_title,
        cic.ed_cicle_title,
        room.ed_classroom_title,
        fac.ed_faculty_title
      FROM eduall_students s
      INNER JOIN eduall_class c 
          ON c.ed_class_id = s.ed_student_class AND COALESCE(c.ed_class_deleted,0)=0
      INNER JOIN eduall_academic_year y 
          ON y.ed_academic_year_id = c.ed_class_year AND COALESCE(y.ed_academic_year_deleted,0)=0
      LEFT JOIN eduall_classrooms room 
          ON room.ed_classroom_id = c.ed_class_room AND COALESCE(room.ed_classroom_deleted,0)=0
      LEFT JOIN eduall_courses crs 
          ON crs.ed_course_id = c.ed_class_course AND COALESCE(crs.ed_course_deleted,0)=0
      LEFT JOIN eduall_academic_level lvl 
          ON lvl.ed_academic_level_id = c.ed_class_academic_level AND COALESCE(lvl.ed_academic_level_deleted,0)=0
      LEFT JOIN eduall_cicles cic 
          ON cic.ed_cicle_id = c.ed_class_cicle AND COALESCE(cic.ed_cicle_deleted,0)=0
      LEFT JOIN eduall_faculty fac 
          ON fac.ed_faculty_id = c.ed_class_faculty AND COALESCE(fac.ed_faculty_deleted,0)=0
      WHERE 
        s.ed_student_id = ? 
        AND s.ed_student_institute_code = ?
        AND COALESCE(s.ed_student_deleted,0) = 0
      LIMIT 1;
    `;

    const [student] = await ExecuteQuery(res, studentQuery, [studentCode, institute], true, {
      db: "eduall_students", type: "select", data: req
    });

    if (!student) {
      
      return res.status(404).json({ success: false, message: "Aluno não encontrado." });
    }

    // === Buscar pagamentos do aluno diretamente (apenas colunas necessárias) ===
    const paymentsQuery = `
      SELECT ed_fee_payment_month
      FROM eduall_fees_payments
      WHERE ed_fee_payment_deleted = 0
        AND ed_fee_payment_student = ?
        AND ed_fee_payment_institute_code = ?
        AND ed_fee_payment_academic_year = ?;
    `;

    const payments = await ExecuteQuery(res, paymentsQuery, [student.ed_student_id, institute, ano], true, {
      db: "eduall_fees_payments", type: "select", data: req
    });

    // Transformar pagamentos em um Set (busca O(1))
    const paidMonthsSet = new Set(payments.map(p => parseFloat(p.ed_fee_payment_month)));

    // === Gerar meses de forma mais direta (sem loops desnecessários) ===
    const MonthsList = [
      "Janeiro","Fevereiro","Março","Abril","Maio","Junho",
      "Julho","Agosto","Setembro","Outubro","Novembro","Dezembro"
    ];

    const buildMonths = (start, end) => {
      if (!start || !end || !start.includes("-") || !end.includes("-")) return [];
      const [Y1, M1] = start.split("-").map(Number);
      const [Y2, M2] = end.split("-").map(Number);

      const months = [];
      const pushMonth = (m, y) => months.push({ 
        month: `${MonthsList[m]} de ${y}`, 
        code: parseFloat(`${m}.${y}`) 
      });

      if (Y1 === Y2) {
        for (let m = M1 - 1; m <= M2 - 1; m++) pushMonth(m, Y1);
      } else {
        for (let m = M1 - 1; m <= 11; m++) pushMonth(m, Y1);
        for (let m = 0; m <= M2 - 1; m++) pushMonth(m, Y2);
      }

      return months;
    };

    const months = buildMonths(student.ed_academic_year_startDate, student.ed_academic_year_endDate);
    const now = new Date();
    const curYear = now.getFullYear();
    const curMonth = now.getMonth();

    // === Determinar status de cada mês ===
    const monthsStatus = months.map(m => {
      const paid = paidMonthsSet.has(m.code);
      const [monthIndex, year] = m.code.toString().split(".").map(Number);

      let status = "other";
      if (paid) status = "paid";
      else if (year < curYear || (year === curYear && monthIndex < curMonth)) status = "notpaid";

      return { ...m, status };
    });

    // === Resultado final ===
    const result = {
      student: {
        id: student.ed_student_id,
        code: student.ed_student_code,
        name: student.ed_student_name,
        picture: student.ed_student_picture,
        class_title: student.ed_class_title,
        course_title: student.ed_course_title,
        classroom_title: student.ed_classroom_title,
        level_title: student.ed_academic_level_title,
        cicle_title: student.ed_cicle_title,
        faculty_title: student.ed_faculty_title,
        year_title: student.ed_academic_year_title,
        academic_year: ano
      },
      months: monthsStatus
    };

    
    return res.status(200).json({ success: true, data: result });

  } catch (error) {
    console.error("❌ ERRO em GetSingleStudentPaymentStatus:", error);
    
    return res.status(500).json({
      success: false,
      message: "Erro ao carregar estado de pagamento do aluno!",
      error: error.message || error
    });
  }
};

function FeespaymentsTable() {
  const [data, setData] = useState([]);
  const [LoadStatus, setLoadStatus] = useState(null);

  function computeLine(item) {
    const price = item.ed_fee_payment_price * 1;
    const discountPct = item.ed_fee_payment_discount * 1 || 0;
    const ivaPct = item.ed_fee_payment_iva * 1 || 0;
    const fineType = item.ed_fee_payment_fineType * 1 || 0;
    const fineValue = item.ed_fee_payment_fineValue * 1 || 0;

    const afterDiscount = price - (price * discountPct) / 100;
    const base = ivaPct > 0 ? afterDiscount / (1 + ivaPct / 100) : afterDiscount;
    const ivaAmount = afterDiscount - base;
    const fineAmount = fineType === 1 ? fineValue : (base * fineValue) / 100;
    const total = afterDiscount + fineAmount;

    return { base, ivaAmount, fineAmount, total, afterDiscount };
  }

  async function loadData() {
    try {
      setLoadStatus(0);

      const response = await axios.get(TABLEURL[0]); // API já retorna .extras
      const grouped = {};

      (response.data || []).forEach((item) => {
        const { base, ivaAmount, fineAmount, total, afterDiscount } = computeLine(item);
        const key = item.ed_fee_payment_invoiceCode;

        const deleteUrl = Hoot() + `eduallfeepaymentsdelete/delete/${item.ed_fee_payment_id}`;
        const fromServiceFlag = (item.ed_fee_payment_from_service * 1) === 1;

        const paymentLine = {
          id: item.ed_fee_payment_id,
          month: item.ed_fee_payment_month,
          service: item.ed_service_title,
          price: item.ed_fee_payment_price * 1,
          discount: item.ed_fee_payment_discount * 1,
          iva: item.ed_fee_payment_iva * 1,
          fineType: item.ed_fee_payment_fineType * 1,
          fineValue: item.ed_fee_payment_fineValue * 1,
          method: item.ed_fee_payment_type * 1,
          registerDate: item.ed_fee_payment_registerDate,
          coin: item.ed_coin_code,
          base,
          ivaAmount,
          fineAmount,
          total,
          afterDiscount,
          paymentCode: item.ed_fee_payment_code,
          invoiceFile: item.ed_fee_payment_invoiceFile,
          fromService: fromServiceFlag
        };

        const extrasArray = item.extras ? JSON.parse(item.extras) : [];
        const extrasTotal = Array.isArray(extrasArray)
          ? extrasArray.reduce((sum, ex) => sum + (Number(ex.amount) || 0), 0)
          : 0;

        if (!grouped[key]) {
          grouped[key] = {
            id: item.ed_fee_payment_id,
            service: item.ed_service_title,
            coin: item.ed_coin_code,
            code: item.ed_fee_payment_invoiceFile || item.ed_service_payment_invoiceCode,
            paymentCode: item.ed_fee_payment_code,
            student: item.ed_fee_payment_student,
            name: item.ed_student_name,
            studentcode: item.ed_student_code,
            picture: item.ed_student_picture,
            months: [item.ed_fee_payment_month],
            balance: total,
            extrasTotal,
            price: item.ed_fee_payment_price,
            invoiceN: item.ed_fee_payment_invoiceCode,
            date: moment(item.ed_fee_payment_registerDate).format('DD [de] MMMM [de] YYYY [às] HH:mm'),
            invoiceCount: 1,
            paymentsUrls: [deleteUrl],
            payments: [paymentLine],
            extras: extrasArray,
            extrasAdded: false,
            fromService: fromServiceFlag   
          };
        } else {
          grouped[key].balance += total;
          grouped[key].months.push(item.ed_fee_payment_month);
          grouped[key].invoiceCount += 1;
          grouped[key].paymentsUrls.push(deleteUrl);
          grouped[key].payments.push(paymentLine);

          // se qualquer linha veio dos serviços, marcamos toda a fatura como true
          if (fromServiceFlag) {
            grouped[key].fromService = true;
          }
        }

        if (!grouped[key].extrasAdded && extrasTotal > 0) {
          grouped[key].balance += extrasTotal;
          grouped[key].extrasTotal = extrasTotal;
          grouped[key].extras = extrasArray;

        }
      });

      const rows = Object.values(grouped).map((g, index) => ({

            let mm, yy, monthName;

              const monthIndex = parseInt(mm, 10);

                  ? Months[monthIndex]

            } else if (String(m).includes("-")) {

          .filter(Boolean)

        fromService: !!g.fromService

const FineBadgeUnified = React.memo(function FineBadgeUnified({
  studentId,
  paymentType,
  servicePrice,
  selectedMonthCodes,
  onFineLoaded,
}) {
  const [phase,        setPhase]        = useState("idle");   // idle | loading | done | error
  const [fineData,     setFineData]     = useState(null);     // resposta completa do backend
  const [rules,        setRules]        = useState([]);
  const [selectedRule, setSelectedRule] = useState(null);
  const [rulesPhase,   setRulesPhase]   = useState("idle");   // idle | loading | done | error

  const abortRef       = useRef(null);
  const onFineLoadedRef = useRef(onFineLoaded);
  useEffect(() => { onFineLoadedRef.current = onFineLoaded; }, [onFineLoaded]);

  const URL_DETECT   = useMemo(() => Hoot() + "eduallmultascalculate/student/post/", []);
  const URL_CALC     = useMemo(() => Hoot() + "eduallmultascalculatebyrule/student/post/", []);
  const URL_RULES    = useMemo(() => Hoot() + "eduallmultasrules/get/", []);

  // ── Carrega regras disponíveis (uma vez por paymentType) ──────────────────
  useEffect(() => {
    if (!studentId) return;
    let cancelled = false;
    setRulesPhase("loading");
    axios.get(URL_RULES)
      .then(r => {
        if (cancelled) return;
        const all = r.data || [];
        const isStr = paymentType && typeof paymentType === "string"
          && paymentType.trim() !== "" && isNaN(Number(paymentType));
        const active = all.filter(rule => {
          if (!rule.fine_rule_active) return false;
          if (isStr) return rule.fine_rule_payment_type === paymentType;
          return true;
        });
        setRules(active);
        setRulesPhase("done");
      })
      .catch(() => { if (!cancelled) setRulesPhase("error"); });
    return () => { cancelled = true; };

  // ── Calcula multa via backend ─────────────────────────────────────────────
  const calculate = useCallback(async (rule, codes) => {
    if (!studentId || !rule) return;

    // Cancela pedido anterior
    if (abortRef.current) abortRef.current.abort();
    const controller = new AbortController();
    abortRef.current = controller;

    setPhase("loading");
    setFineData(null);

    try {
      const { data } = await axios.post(URL_CALC, {
        studentId,
        ruleId:             rule.fine_rule_id,
        selectedMonthCodes: codes.map(c => String(c)),
        paymentType:        paymentType || "propina",
        servicePrice:       Number(servicePrice || 0),
      }, { signal: controller.signal });

      setFineData(data);
      setPhase("done");

      const totalFine = Number(data?.totalFine || 0);
      onFineLoadedRef.current?.({
        hasFine:   totalFine > 0,
        fineValue: totalFine,
        reason:    rule.fine_rule_name,
        currency:  data?.currency || "",
        explicacao_html: data?.explicacao_html || null,
      });

    } catch (err) {
      if (axios.isCancel(err)) return;
      setPhase("error");
      // fallback com valor fixo da regra
      const fallback = Number(rule.fine_rule_fixed_value || 0) * Math.max(1, codes.length);
      onFineLoadedRef.current?.({
        hasFine: fallback > 0, fineValue: fallback,
        reason: rule.fine_rule_name, currency: "", explicacao_html: null,
      });
    }
  }, [studentId, paymentType, servicePrice, URL_CALC]);

  useEffect(() => {

        : null;
      const rule     = match || rules[0];
      setSelectedRule(rule);
      if (selectedMonthCodes && selectedMonthCodes.length > 0) {
        calculate(rule, selectedMonthCodes);
      } else {
        setPhase("idle");
        onFineLoadedRef.current?.({ hasFine: false, fineValue: 0 });
      }
    }).catch(() => {
      if (cancelled) return;
      // Sem detecção — usa primeira regra se tiver meses
      const rule = rules[0];
      setSelectedRule(rule);
      if (selectedMonthCodes && selectedMonthCodes.length > 0) {
        calculate(rule, selectedMonthCodes);
      } else {
        setPhase("idle");
      }
    });
    return () => { cancelled = true; };
  }, [studentId, rulesPhase, rules]); // eslint-disable-line

  useEffect(() => {

  if (!studentId) return null;

        <_FBSpinnerEl /> A calcular multa automática…

function BalanceCreditSelector({ balance, total, coin, editAmount, onChange }) {
  if (balance === null || balance === undefined) return null;
  const maxUsable   = Math.min(Number(balance), Number(total));
  const current     = Number(editAmount ?? 0);
  const remaining   = Math.max(0, Number(total) - current);
  const balanceLeft = Number(balance) - current;
  const fullyPaid   = remaining <= 0;
  return (
    <CreditSelectorBox fullyPaid={fullyPaid}>
      <div className="cs-header">
        <span className="cs-title"><IoWalletOutline size={14} /> Créditos a usar neste pagamento</span>
        <span className="cs-available">Disponível: <strong>{fmtMoney(balance, coin)}</strong></span>
      </div>
      <input type="range" min={0} max={maxUsable} step={1} value={current}
        onChange={e => onChange(Number(e.target.value))} className="cs-slider" />
      <div className="cs-input-row">
        <div className="cs-input-wrap">
          <input type="number" min={0} max={maxUsable} step={1} value={current}
            onChange={e => onChange(Math.min(Math.max(0, Number(e.target.value)), maxUsable))}
            className="cs-input" placeholder="0" />
          <span className="cs-coin">{coin}</span>
        </div>
        <div className="cs-quick-btns">
          <button type="button" className="cs-btn cs-btn-all"  onClick={() => onChange(maxUsable)}>Usar tudo</button>
          <button type="button" className="cs-btn cs-btn-half" onClick={() => onChange(Math.floor(maxUsable / 2))}>Metade</button>
          <button type="button" className="cs-btn cs-btn-none" onClick={() => onChange(0)}>Não usar</button>
        </div>
      </div>
      <div className={`cs-summary ${fullyPaid ? "cs-summary--ok" : "cs-summary--partial"}`}>
        <div className="cs-summary-item">
          <div className="cs-summary-label">Créditos usados</div>
          <div className="cs-summary-value cs-purple">{fmtMoney(current, coin)}</div>
        </div>
        <div className="cs-summary-divider" />
        <div className="cs-summary-item">
          <div className="cs-summary-label">Saldo após pagamento</div>
          <div className={`cs-summary-value ${balanceLeft < 0 ? "cs-red" : "cs-green"}`}>{fmtMoney(balanceLeft, coin)}</div>
        </div>
        <div className="cs-summary-divider" />
        <div className="cs-summary-item">
          <div className="cs-summary-label">Ainda a pagar</div>
          <div className={`cs-summary-value ${remaining > 0 ? "cs-orange" : "cs-green"}`}>
            {remaining > 0

              : <><IoCheckmarkCircle   size={12} style={{ marginRight: 3 }} />Coberto</>}
          </div>
        </div>
      </div>
      {remaining > 0 && current > 0 && (
        <div className="cs-warning"><strong>Pagamento parcial</strong><br />
          Faltam <strong>{fmtMoney(remaining, coin)}</strong> — cobrar por outro método.</div>
      )}
      {current === 0 && (
        <div className="cs-info">Sem créditos aplicados. O valor total será cobrado por outro método.</div>
      )}
      {fullyPaid && current > 0 && (

      )}
    </CreditSelectorBox>
  );
}

  async function prepareMonths(studentId) {

      const [Cls] = await Promise.all([axios.get(URL.singleClass + S.ed_student_class), null]);
      const C = Cls?.data?.[0];

      const AY = await axios.get(URL.singleAcademicYear + C.ed_class_year);
      const Y  = AY?.data?.[0];
      if (!Y) return;
      setField("feepayment_academic_year", C.ed_class_year);
      setCurrentAcademicYear(C.ed_class_year);
      const [y1, m1] = Y.ed_academic_year_startDate.split("-").map(v => v * 1);
      const [y2, m2] = Y.ed_academic_year_endDate.split("-").map(v => v * 1);
      const opts = [];
      const pushMonth = (m, y) => opts.push({ label: `${MonthsList[m]} de ${y}`, selected: false, value: Number(`${m}.${y}`) });
      if (y2 - y1 === 1 && m2 <= m1) {
        for (let m = m1 - 1; m <= 11; m++) pushMonth(m, y1);
        for (let m = 0; m <= m2 - 1; m++) pushMonth(m, y2);
      } else if (y2 - y1 === 1 && m2 > m1) {
        for (let m = m1 - 1; m <= 11; m++) pushMonth(m, y1);
        for (let m = 0; m <= m2 - 1; m++) pushMonth(m, y2);
      } else {
        for (let m = m1 - 1; m <= m2 - 1; m++) pushMonth(m, y1);
      }
      setMonthOptions(opts);
      if (C.ed_class_service) {
        setServiceCode(C.ed_class_service);
        setField("feepayment_servicecode", C.ed_class_service);
        await loadServiceData(C.ed_class_service, 1, studentId);
      }
    } catch { toast.error("Erro ao recolher meses do ano letivo."); }

      finePaymentTypeRef.current = newType;
      setFinePaymentType(newType);

      const instIVA = respIVA?.data?.[0]?.ed_institute_iva_value * 1 || 0;
      setServiceIva(S.ed_service_iva * 1 > 0 ? instIVA : 0);

      try {
        const P  = respParents?.data || [];

        const ST = st?.data?.[0];
        const filtered = [];
        if (ST) {
          const codeSet = new Set([ST.ed_student_code]);
          P.forEach(p => {

            if (arr.some(c => codeSet.has(c))) filtered.push(p);
          });

        setParentsData(filtered);
      } catch { setParentsData([]); }

      recalcPreview(amount || selectedMonths.length, S);
    } catch {}

  const recalcPreview = useCallback((amount, SVC) => {

    const DISCOUNT = (SVC?.ed_service_discount ?? serviceDiscount) * 1 || 0;
    const IVA_ON   = (SVC?.ed_service_iva      ?? (serviceIva > 0 ? 1 : 0)) * 1;
    const IVA_PCT  = serviceIva * 1 || 0;
    const discountAmount     = DISCOUNT > 0 ? (PRICE * DISCOUNT) / 100 : 0;
    const priceAfterDiscount = Math.max(0, PRICE - discountAmount);
    const extrasTotal        = totalExtraServices();
    const ivaAmount          = IVA_ON ? (priceAfterDiscount * IVA_PCT) / 100 : 0;
    const lineTotal          = priceAfterDiscount + ivaAmount;
    const grand              = lineTotal * Math.max(1, amount || 1) + extrasTotal;
    setInvoiceValue(NumberToPrice(grand) + " " + (serviceCoin || ""));
    setBalancePreview(grand);
    setApplyDiscount(DISCOUNT > 0);
    setBalanceEditAmount(prev => {
      if (prev === null) return null;
      return Math.min(prev, Math.min(studentBalance ?? 0, grand));

              {/* MESES */}
              {!props.update ? (
                monthOptions.length ? (
                  <Form.Group className="mb-3">
                    <Form.Label>Mês(es) a pagar <span className="text-danger ml-2">*</span></Form.Label>
                    {lockMonths ? (
                      <div className="grey-text" style={{ fontSize: 13 }}>
                        Meses: <span className="text-main">{selectedMonths.map(m => m.label).join(", ")}</span>

                          onChange={onChangeMonths} labelledBy="select" />
                        {errors.feepayment_months && <small className="text-danger">{errors.feepayment_months}</small>}
                      </>
                    )}
                  </Form.Group>
                ) : null
              ) : (
                <Form.Group className="mb-3">
                  <Form.Label>Mês a pagar <span className="text-danger ml-2">*</span></Form.Label>
                  <Form.Select disabled id="feepayment_months" value={form.feepayment_months || currentMonth || ""}>
                    {monthOptions.map((m, i) => (
                      <option key={i} value={m.value}
                        selected={Number(m.value) === Number(currentMonth)}>{m.label}</option>
                    ))}
                  </Form.Select>
                </Form.Group>
              )}

              {!studentCode && !props.update && (
                <NoStudentPlaceholder>
                  <span style={{ fontSize: 22 }}>👤</span>
                  <div>
                    <div style={{ fontWeight: 600, color: "#374151", fontSize: 14 }}>Seleccione um aluno para continuar</div>
                    <div style={{ color: "#9ca3af", fontSize: 12, marginTop: 3 }}>
                      Os meses, serviço e condições de pagamento carregam automaticamente após seleccionar o aluno.
                    </div>
                  </div>
                </NoStudentPlaceholder>
              )}

              {(studentCode || props.update) && (<>

              {/* SERVIÇO */}
              <Form.Group className="mb-3">
                <Form.Label>
                  Valor associado à propina <span className="text-danger ml-2">*</span>
                  {serviceCode && <AutoBadge>preenchido automaticamente</AutoBadge>}
                </Form.Label>
                {servicesLoading ? (
                  <ServiceLoadingBox><_FBSpinnerEl /> A carregar serviços…</ServiceLoadingBox>
                ) : (
                  <div className={`select-search-item ${errors.feepayment_servicecode ? "error" : ""}`}>
                    <SelectSearch
                      onChange={onSelectService}
                      value={serviceCode ? String(serviceCode) : ""}
                      options={servicesList.map(s => ({ ...s, value: String(s.value) }))}
                      search
                      placeholder="Pesquisar serviço / propina…"
                    />
                  </div>
                )}
                <small className="text-danger">{errors.feepayment_servicecode}</small>
                {serviceCode != null && (
                  <div className="mt-2">
                    <Badge className="bg-main">
                      <div className="ed-flex">
                        Valor do serviço:
                        <div className="ml-1"><GetServicePrice ID={serviceCode} /></div>
                      </div>
                    </Badge>
                  </div>
                )}
              </Form.Group>

              {/* MULTA INCLUÍDA NA FATURA */}
              {extraServices.some(it => it._isFine) && (
                <FineItemsBox>
                  <FineItemsTitle>
                    <MdGavel size={13} color="#b91c1c" />
                    <span>Multa incluída nesta fatura</span>
                  </FineItemsTitle>
                  {extraServices.filter(it => it._isFine).map((it, idx) => (
                    <FineItemRow key={idx}>
                      <FineItemName>{it.name || "Multa automática"}</FineItemName>
                      <FineItemValue>{_fmtFine(Number(it.value) || 0, serviceCoin)}</FineItemValue>
                      <FineLockedBadge title="Calculado automaticamente">automático</FineLockedBadge>
                    </FineItemRow>
                  ))}
                </FineItemsBox>
              )}

              {/* VALORES ADICIONAIS MANUAIS */}
              <div className="mt-2">
                <FormControlLabel
                  control={<Checkbox checked={extraServicesEnabled} onChange={e => setExtraServicesEnabled(e.target.checked)} />}
                  label="Adicionar valores adicionais manuais" />
              </div>
              {extraServicesEnabled && (
                <BoxContainerBorder className="boxItem">
                  <h2 className="title">Valores adicionais</h2>
                  {extraServices.filter(it => !it._isFine).length === 0 && (
                    <div className="mb-3"><small className="grey-text">Sem itens. Clique em <strong>"+ Adicionar"</strong> para criar uma linha.</small></div>
                  )}
                  {extraServices.filter(it => !it._isFine).map(it => {
                    const realIdx = extraServices.indexOf(it);
                    return (
                      <div key={realIdx} className="ed-space mb-2">
                        <div className="ed-flex" style={{ gap: 10, flex: 1 }}>
                          <Form.Control type="text" placeholder="Nome (ex: Seguro escolar)" value={it.name}
                            onChange={e => updateExtraService(realIdx, "name", e.target.value)} />
                          <Form.Control type="number" step="0.01" min="0"
                            placeholder={`Valor (${serviceCoin || "KZ"})`} value={it.value}
                            onChange={e => updateExtraService(realIdx, "value", e.target.value)}
                            style={{ maxWidth: 180 }} />
                        </div>
                        <div onClick={() => removeExtraService(realIdx)} className="bg-red btn ml-2">
                          <DeleteOutline fontSize="small" /> Remover
                        </div>
                      </div>
                    );
                  })}
                  <div className="mt-3">
                    <div className="btn btn-bordered bg-main text-main" onClick={addExtraService}>
                      <AddCircleOutlineIcon fontSize="small" /> Adicionar
                    </div>
                  </div>
                </BoxContainerBorder>
              )}

              {/* CHECKBOX MULTA AUTOMÁTICA */}
              <div className="mt-2 mb-1">
                <FormControlLabel
                  control={
                    <Checkbox
                      checked={applyFineManual}
                      onChange={e => {
                        const checked = e.target.checked;
                        setApplyFineManual(checked);
                        if (!checked) {
                          // Desmarcou — limpa multa imediatamente
                          setExtraServices(prev => prev.filter(ex => !ex._isFine));
                        }
                      }}
                      size="small"
                    />
                  }
                  label={<span style={{ fontSize: 13, color: "#374151" }}>Verificar e aplicar multa automática (se existir)</span>}
                />
              </div>

              {/* v29: UM ÚNICO BADGE — aparece apenas quando checkbox está marcada */}
              {shouldShowFineBadge && (
                <FinesBadgeWrapper>
                  <FineBadgeUnified
                    key={fineBadgeKey}
                    studentId={studentCode}
                    paymentType={finePaymentType}
                    servicePrice={Number(servicePrice || 0)}
                    selectedMonthCodes={selectedMonthCodes}
                    onFineLoaded={handleFineLoaded}
                  />
                </FinesBadgeWrapper>
              )}

              {/* MÉTODO DE PAGAMENTO */}
              <Form.Group>
                <div className="mt-2">
                  <div className="block">
                    <Form.Label>Método de pagamento <span className="text-danger ml-2">*</span></Form.Label>
                    <Form.Select onChange={onChangeInput}
                      className={!!errors.feepayment_type && "is-invalid"}
                      value={paymentType ?? ""} isInvalid={!!errors.feepayment_type}
                      id="feepayment_type" required>
                      <option disabled value="">Selecionar método de pagamento...</option>
                      {PAYMENT_METHODS.map(m => (
                        <option key={m.value} value={m.value}
                          disabled={m.value === PAYMENT_TYPE_BALANCE && (!studentCode || studentBalance === null)}>
                          {m.value === PAYMENT_TYPE_BALANCE && studentBalance !== null
                            ? `${m.label} (${fmtMoney(studentBalance, studentBalanceCoin)})` : m.label}
                        </option>
                      ))}
                    </Form.Select>
                    <Form.Control.Feedback type="invalid">{errors.feepayment_type}</Form.Control.Feedback>
                  </div>
                  {usingBalance && studentCode && studentBalance !== null && (
                    <BalanceCreditSelector
                      balance={studentBalance} total={balancePreview}
                      coin={studentBalanceCoin || serviceCoin}
                      editAmount={balanceEditAmount} onChange={setBalanceEditAmount}
                    />
                  )}
                  {paymentType !== null && paymentType !== 0 && paymentType !== PAYMENT_TYPE_BALANCE && (
                    <div className="block mt-3">
                      <Form.Label>Nº de Referência</Form.Label>
                      <Form.Control type="text" onChange={onChangeInput}
                        className={!!errors.feepayment_bordereux_number && "is-invalid"}
                        value={form.feepayment_bordereux_number || ""}
                        isInvalid={!!errors.feepayment_bordereux_number}
                        id="feepayment_bordereux_number" />
                      <Form.Control.Feedback type="invalid">{errors.feepayment_bordereux_number}</Form.Control.Feedback>
                      <div className="block mt-3">
                        <Form.Label>Nome do banco</Form.Label>
                        <Form.Control type="text" onChange={onChangeInput}
                          className={!!errors.feepayment_bank && "is-invalid"}
                          value={form.feepayment_bank || ""}
                          isInvalid={!!errors.feepayment_bank} id="feepayment_bank" />
                        <Form.Control.Feedback type="invalid">{errors.feepayment_bank}</Form.Control.Feedback>
                      </div>
                      <div className="block mt-3">
                        <Form.Label>Local de pagamento</Form.Label>
                        <Form.Control type="text" onChange={onChangeInput}
                          className={!!errors.feepayment_place && "is-invalid"}
                          value={form.feepayment_place || ""}
                          isInvalid={!!errors.feepayment_place} id="feepayment_place" />
                        <Form.Control.Feedback type="invalid">{errors.feepayment_place}</Form.Control.Feedback>
                      </div>
                    </div>
                  )}
                </div>
              </Form.Group>
              </>)}

              {/* FICHEIROS */}
              {(studentCode || props.update) && currentFiles.length > 0 && (
                <BoxContainerBorder className="boxItem mt-4">
                  <h2 className="title">Ficheiros anexados</h2>
                  <ul className="student-file-cp">
                    {currentFiles.map((item, index) => (
                      <li className="student-file-el" id={`student-file-el-${index}`} key={index}>
                        <div className="ed-space">
                          <div className="ed-flex">
                            <div className="icon"><HiOutlineDocumentText /></div>
                            <div className="block">
                              <div className="title text-main">
                                <ReduceTextLength text={item.ed_file_name} length={window?.width >= 1300 ? 80 : 50} />
                              </div>
                              <div className="ed-flex">
                                <div className="date text-dark"><AccessTimeOutlined /><span>{item.ed_file_register_date}</span></div>
                                | <div className="size text-dark">{NumberToBytes(item.ed_file_size)}</div>
                              </div>
                            </div>
                          </div>
                          <div className="delete">
                            <DeleteModal remove_element={`#student-file-el-${index}`} title="este ficheiro"
                              url={Hoot() + `eduallfiledelete/delete/${item.ed_file_id}`}
                              message="Ficheiro eliminado com sucesso"
                              toggle_btn={<div className="btn-circle bg-danger ml-2 text-light"><DeleteOutline /></div>} />
                          </div>
                        </div>
                      </li>
                    ))}
                  </ul>
                </BoxContainerBorder>
              )}

              {(studentCode || props.update) && (
                <div className="mt-2">
                  <FormControlLabel onChange={showBoxFile} control={<Checkbox />} label="Anexar recibos de pagamento" />
                </div>
              )}
              {(studentCode || props.update) && (
                <BoxContainerBorder className={docfile ? "boxItem" : "boxItem d-none"}>
                  <h2 className="title">Carregar Recibo de pagamento</h2>
                  <FileUpload input_name="feepayment_files" Icon="0"
                    type_of_files="image/x-png,image/gif,image/jpeg,Docs/pdf,application/vnd.openxmlformats-officedocument.wordprocessingml.document, application/pdf"
                    extensions="pdf, docx, png, jpeg, jpg" Files={GetFiles} ref={ChildRef} />
                </BoxContainerBorder>
              )}
              {docfile && <br />}
            </Modal.Body>

            {/* FOOTER */}
            <Modal.Footer>
              <div className="ed-space">
                <div className="ed-wrap">
                  <TX className="mt-1">
                    <h4>Desconto : ({serviceDiscount})%</h4>
                    {applyDiscount ? <div className="yes"><Check /></div> : <div className="no"><Close /></div>}
                  </TX>
                  <TX className="mt-1">
                    <h4 className="ml-2">IVA : ({serviceIva})%</h4>
                    {serviceIva * 1 > 0 ? <div className="yes"><Check /></div> : <div className="no"><Close /></div>}
                  </TX>
                </div>
              </div>
              {studentCode && studentBalance !== null && (
                <BalanceFooterRow>
                  <IoWalletOutline size={15} />
                  <span>Saldo do aluno:</span>
                  <strong style={{ color: Number(studentBalance) < 500 ? "#e21f1f" : "var(--ed-green-light, #15803d)" }}>
                    {fmtMoney(studentBalance, studentBalanceCoin || serviceCoin)}
                  </strong>
                  {usingBalance && balanceEditAmount !== null && balanceEditAmount > 0 && balanceEditAmount < studentBalance && (
                    <span style={{ fontSize: 11, color: "#7c3aed", opacity: .8 }}>
                      (usando {fmtMoney(balanceEditAmount, studentBalanceCoin || serviceCoin)})
                    </span>
                  )}
                  {usingBalance && balancePreview > 0 && balanceEditAmount !== null && balanceEditAmount > 0 && (
                    <>
                      <span style={{ opacity: .5 }}>&rarr;</span>
                      <strong style={{ color: "var(--ed-green-light, #15803d)" }}>
                        {fmtMoney(Number(studentBalance) - balanceEditAmount, studentBalanceCoin || serviceCoin)}
                      </strong>
                    </>
                  )}
                </BalanceFooterRow>
              )}
              <div className="ed-space">
                <div className="ed-flex">
                  <h5 className="m-0">Total :</h5>
                  <Form.Control readOnly disabled value={invoiceValue} id="feepayment_balance"
                    className="text-danger ml-2" style={{ maxWidth: "210px" }} />
                </div>
                <div className="ed-flex">
                  <Button className="bg-light text-dark" onClick={handleClose}>Cancelar</Button>
                  <SaveButton
                    icon={props.title ? <Update /> : <MonetizationOn />}
                    title={props.title ? props.title : "Pagar"}
                    status={registerStatus}
                    disabled={usingBalance && studentBalance !== null && (balanceEditAmount ?? 0) > studentBalance}
                  />
                </div>
              </div>
            </Modal.Footer>
          </Form>
        </Modal>
      } />

      {studentCode && (
        <StudentBalanceModal mode="modal" canvas={true} show={showBalanceCanvas}
          onHide={() => { setShowBalanceCanvas(false); fetchStudentBalance(studentCode); }}
          studentId={studentCode} currency={studentBalanceCoin || serviceCoin || "KZ"} />
      )}
    </div>
  );
}
