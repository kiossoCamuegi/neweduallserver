// image cache for PDF
const _imgCache = {};
async function imgFetch(src) {
  if (!src||_imgCache[src]===null) return null;
  if (_imgCache[src]) return _imgCache[src];
  try {
    const blob = await (await fetch(src,{mode:"cors"})).blob();
    return new Promise(ok=>{
      const fr=new FileReader();
      fr.onload=()=>{_imgCache[src]=fr.result;ok(fr.result);};
      fr.readAsDataURL(blob);
    });
  } catch { _imgCache[src]=null; return null; }
}

function StudentMultiSelect({ options, value, onChange }) {
  const [search, setSearch] = React.useState("");
  const filtered    = options.filter(o=>o.label.toLowerCase().includes(search.toLowerCase()));
  const selectedSet = new Set(value.map(v=>v.value));
  const toggle = (opt) => { if (selectedSet.has(opt.value)) onChange(value.filter(v=>v.value!==opt.value)); else onChange([...value,opt]); };
  const toggleAll = () => {
    if (filtered.every(o=>selectedSet.has(o.value))) { const fs=new Set(filtered.map(f=>f.value)); onChange(value.filter(v=>!fs.has(v.value))); }
    else { const m=[...value]; filtered.forEach(o=>{if(!selectedSet.has(o.value))m.push(o);}); onChange(m); }
  };
  const allSel = filtered.length>0 && filtered.every(o=>selectedSet.has(o.value));
  return (
    <div style={{fontFamily:F_SANS}}>

        <input type="text" value={search} onChange={e=>setSearch(e.target.value)} placeholder="Pesquisar aluno…"

// ─────────────────────────────────────────────────────────────────────────────
// UNIFIED LANDSCAPE PDF ENGINE — Modelo Normal + Modelo Exame
// Both: A4 landscape, all disciplines as columns
//
// model="normal"  → sub-cols: MT1 | MT2 | MT3 | MFD  (configurable, removable)
//                   sigs: Professor | Sub-Dir Pedagógico | Director
// model="exame"   → sub-cols: MFD | NE | MF | REC     (configurable, removable)
//                   sigs: Corpo de Júri | Sub-Dir Pedagógico | Director da Escola
// hiddenCols = Set of col keys to skip (e.g. "MFD","NE","REC")
// ─────────────────────────────────────────────────────────────────────────────

// Default sub-column definitions per model
const MODEL_COLS = {
  normal: [
    {key:"MT1", label:"MT1", getValue:(g)=>{ const t=g?.terms?.find(t=>t.quarter===1)||{}; return t.mt??null; }},
    {key:"MT2", label:"MT2", getValue:(g)=>{ const t=g?.terms?.find(t=>t.quarter===2)||{}; return t.mt??null; }},
    {key:"MT3", label:"MT3", getValue:(g)=>{ const t=g?.terms?.find(t=>t.quarter===3)||{}; return t.mt??null; }},
    {key:"MFD", label:"MFD", getValue:(g)=>g?.mfd??null},
  ],
  exame: [
    {key:"MFD", label:"MFD", getValue:(g)=>g?.mfd??null},
    {key:"NE",  label:"NE",  getValue:(g)=>g?.exam_score??null},
    {key:"MF",  label:"MF",  getValue:(g)=>g?.mf??null},
    {key:"REC", label:"REC", getValue:()=>null},  // manual fill
  ],
};

async function exportLandscapePDF(allDatasets, customLogoUrl, model, hiddenCols, headerLines) {
  const BK=[0,0,0];
  const coatSrc=customLogoUrl||ANGOLA_COAT;
  const coatB64=await imgFetch(coatSrc);

  const PW=297,PH=210,M=7;
  const pdf=new jsPDF({orientation:"landscape",unit:"mm",format:"a4",compress:true});

  const allModelCols = MODEL_COLS[model]||MODEL_COLS.normal;
  const activeCols   = allModelCols.filter(c=>!hiddenCols.has(c.key));
  const nSub = activeCols.length;

  for(let di=0;di<allDatasets.length;di++){
    if(di>0) pdf.addPage([PW,PH],"landscape");
    const dataset=allDatasets[di];
    const {school,class:cls,students}=dataset.data;
    const passValue=dataset.pass_value??10;
    const gcR=makeGcR(passValue);
    const schoolName=school?.name||school?.institute_name||"";
    const clsTitle=cls?.class_title||cls?.title||"";
    const clsCode=cls?.class_code||cls?.code||"";
    const clsPeriod=cls?.class_period||cls?.period||"";
    const clsRoom=cls?.classroom?.title||cls?.classroom||"";
    const dirName=cls?.class_director?.name||cls?.director_name||"";
    const teacherName=cls?.class_director?.name||"";
    const allSubjects=students[0]?.grades?.map(g=>({id:g.subject_id,title:g.subject_title,is_key:g.is_key}))||[];

    let y=M;

    // Logo
    if(coatB64){try{pdf.addImage(coatB64,"PNG",PW/2-7,y,14,14);}catch{}}
    y+=16;

    // Header lines
    const defLines=[
      {id:1,text:"REPÚBLICA DE ANGOLA",bold:false,isSchool:false},
      {id:2,text:"MINISTÉRIO DA EDUCAÇÃO",bold:false,isSchool:false},
      {id:3,text:"",bold:true,isSchool:true},
    ];
    const hLines=(headerLines&&headerLines.length)?headerLines:defLines;
    for(const line of hLines){
      const txt=line.isSchool?schoolName.toUpperCase():(line.text||"").toUpperCase();
      if(!txt)continue;
      pdf.setFont("helvetica",line.bold?"bold":"normal");
      pdf.setFontSize(line.bold?7.5:6);
      pdf.setTextColor(...BK);
      pdf.text(txt,PW/2,y,{align:"center",maxWidth:PW-M*2});
      y+=line.bold?4:3;
    }

    // Title
    y+=1;
    const titleText=model==="exame"
      ?"PAUTA PARA AS CLASSES DE EXAME"
      :"PAUTA DE CLASSIFICAÇÃO";
    pdf.setFont("helvetica","bold");pdf.setFontSize(7.5);pdf.setTextColor(...BK);
    pdf.text(titleText,PW/2,y,{align:"center"});
    y+=4;

    // Info line
    pdf.setFont("helvetica","normal");pdf.setFontSize(6);
    const parts=[
      clsCode   ?`TURMA: ${clsCode}`   :"",
      clsTitle  ?`CLASSE: ${clsTitle}` :"",
      clsPeriod ?`PERÍODO: ${clsPeriod}`:"",
      clsRoom   ?`SALA: ${clsRoom}`    :"",
    ].filter(Boolean).join("     ");
    pdf.text(parts,PW/2,y,{align:"center"});
    y+=3;

    // Double separator
    pdf.setDrawColor(...BK);pdf.setLineWidth(0.6);pdf.line(M,y,PW-M,y);y+=0.8;
    pdf.setLineWidth(0.2);pdf.line(M,y,PW-M,y);y+=2;

    // Column widths
    const nW=6,nameW=30,obsW=8;
    const remW=(PW-M*2)-nW-nameW-obsW;
    const dW=remW/allSubjects.length;
    const sW=dW/nSub;

    // Head rows
    const headRow1=[
      {content:"Nº",rowSpan:3,styles:{valign:"middle",halign:"center",fontSize:5,fontStyle:"bold"}},
      {content:"NOME COMPLETO",rowSpan:3,styles:{valign:"middle",halign:"center",fontSize:5,fontStyle:"bold"}},
      {content:"DISCIPLINAS",colSpan:allSubjects.length*nSub,styles:{halign:"center",fontSize:5.5,fontStyle:"bold"}},
      {content:"OBS",rowSpan:3,styles:{valign:"middle",halign:"center",fontSize:4.5,fontStyle:"bold"}},
    ];
    const headRow2=allSubjects.map(s=>({
      content:s.title.length>10?s.title.slice(0,10)+"…":s.title,
      colSpan:nSub,
      styles:{halign:"center",fontSize:4,fontStyle:"bold"},
    }));
    const headRow3=[];
    allSubjects.forEach(()=>{
      activeCols.forEach(c=>{
        headRow3.push({content:c.label,styles:{halign:"center",fontSize:3.8,fontStyle:"bold"}});
      });
    });

    // Body
    const bodyRows=students.map((st,idx)=>{
      const row=[idx+1,st.student_name||""];
      allSubjects.forEach(subj=>{
        const g=st.grades?.find(gg=>gg.subject_id===subj.id);
        activeCols.forEach(c=>{
          const v=c.getValue(g);
          row.push(v!=null?(c.key==="NE"?fmt(v):fmtInt(v)):"");
        });
      });
      row.push("");
      return row;
    });
    const minR=model==="exame"?42:35;
    while(bodyRows.length<minR) bodyRows.push(Array(2+allSubjects.length*nSub+1).fill(""));

    // Col styles
    const colStyles={
      0:{halign:"center",cellWidth:nW},
      1:{halign:"left",cellWidth:nameW,fontSize:4.5},
    };
    let ci=2;
    allSubjects.forEach(()=>{ activeCols.forEach(()=>{colStyles[ci]={halign:"center",cellWidth:sW,fontSize:4.5};ci++;}); });
    colStyles[ci]={halign:"center",cellWidth:obsW};

    autoTable(pdf,{
      startY:y,margin:{left:M,right:M},
      tableWidth:PW-M*2,
      head:[headRow1,headRow2,headRow3],
      body:bodyRows,
      theme:"grid",
      styles:{fontSize:4.5,cellPadding:0.5,lineColor:BK,lineWidth:0.15,textColor:BK,font:"helvetica",overflow:"ellipsize"},
      headStyles:{fillColor:[255,255,255],textColor:BK,fontStyle:"bold",halign:"center",lineWidth:0.2,lineColor:BK},
      columnStyles:colStyles,
      didParseCell(d){
        if(d.section!=="body")return;
        const ri=d.row.index; if(ri>=students.length)return;
        const ci2=d.column.index;
        if(ci2<2||ci2>=2+allSubjects.length*nSub)return;
        const dIdx=Math.floor((ci2-2)/nSub);
        const sIdx=(ci2-2)%nSub;
        const g=students[ri]?.grades?.find(gg=>gg.subject_id===allSubjects[dIdx]?.id);
        const v=activeCols[sIdx]?.getValue(g);
        if(v!=null){d.cell.styles.textColor=gcR(v);d.cell.styles.fontStyle="bold";}
      },
    });

    let ty=pdf.lastAutoTable.finalY+3;

    // Signatures
    if(model==="exame"){
      // Corpo de Júri | Sub-Director | Director da Escola
      pdf.setFont("helvetica","bold");pdf.setFontSize(6);pdf.setTextColor(...BK);
      pdf.text("Corpo de Júri",M,ty);
      pdf.setFont("helvetica","normal");pdf.setFontSize(5.5);
      [1,2,3].forEach((n,i)=>{
        pdf.text(`${n}.`,M,ty+4+i*5);
        pdf.setDrawColor(...BK);pdf.setLineWidth(0.25);
        pdf.line(M+4,ty+4.5+i*5,M+40,ty+4.5+i*5);
      });
      const mid=PW/2;
      pdf.setFont("helvetica","bold");pdf.setFontSize(6);pdf.setTextColor(...BK);
      pdf.text("O Sub-Director Pedagógico",mid,ty,{align:"center"});
      pdf.setDrawColor(...BK);pdf.setLineWidth(0.3);
      pdf.line(mid-35,ty+8,mid+35,ty+8);
      if(dirName){pdf.setFont("helvetica","normal");pdf.setFontSize(5.5);pdf.setTextColor(60,60,60);pdf.text(dirName,mid,ty+12,{align:"center"});}
      pdf.setFont("helvetica","bold");pdf.setFontSize(6);pdf.setTextColor(...BK);
      pdf.text("O Director da Escola",PW-M,ty,{align:"right"});
      pdf.setDrawColor(...BK);pdf.setLineWidth(0.3);
      pdf.line(PW-M-70,ty+8,PW-M,ty+8);
    } else {
      // Professor | Sub-Director Pedagógico | Director
      const col1=PW*0.2, col2=PW/2, col3=PW*0.8;
      pdf.setFont("helvetica","bold");pdf.setFontSize(6);pdf.setTextColor(...BK);
      pdf.text("O (A) Professor (A)",col1,ty,{align:"center"});
      pdf.setDrawColor(...BK);pdf.setLineWidth(0.3);
      pdf.line(col1-30,ty+8,col1+30,ty+8);
      pdf.text("O (A) Subdirector(A) Pedagógico",col2,ty,{align:"center"});
      pdf.line(col2-40,ty+8,col2+40,ty+8);
      pdf.setFont("helvetica","normal");pdf.setFontSize(5.5);pdf.setTextColor(80,80,80);
      pdf.text("    /    /",col2,ty+12,{align:"center"});
      pdf.setFont("helvetica","bold");pdf.setFontSize(6);pdf.setTextColor(...BK);
      pdf.text("O (A) Director (A)",col3,ty,{align:"right"});
      pdf.line(col3-30,ty+8,col3,ty+8);
      pdf.setFont("helvetica","normal");pdf.setFontSize(5.5);pdf.setTextColor(80,80,80);
      pdf.text("    /    /",col3-15,ty+12);
    }

    // Footer
    pdf.setFont("helvetica","italic");pdf.setFontSize(4.5);pdf.setTextColor(160,160,160);
    pdf.text(eduallFooterDate(),PW/2,PH-2.5,{align:"center"});
  }

  pdf.save(`Pauta_${model==="exame"?"Exame":"Normal"}_${moment().format("YYYY-MM-DD_HH-mm")}.pdf`);
}

function HeadersEditor({ headers, onChange, onReset, headerLines, onLinesChange, onLinesReset }) {
  const [open, setOpen] = useState(false);
  const [dragIdx, setDragIdx] = useState(null);
  const set = (k,v) => onChange(prev=>({...prev,[k]:v}));

  const addLine = () => {
    const newId = Math.max(0,...headerLines.map(l=>l.id))+1;
    onLinesChange(prev=>[...prev,{id:newId,text:"",bold:false,isSchool:false}]);
  };
  const removeLine = (id) => onLinesChange(prev=>prev.filter(l=>l.id!==id));
  const updateLine = (id,field,val) => onLinesChange(prev=>prev.map(l=>l.id===id?{...l,[field]:val}:l));
  const moveLine = (fromIdx,toIdx) => {
    onLinesChange(prev=>{
      const arr=[...prev];
      const [item]=arr.splice(fromIdx,1);
      arr.splice(toIdx,0,item);
      return arr;
    });
  };

  const inp = (key, placeholder) => (
    <input
      type="text" value={headers[key]} placeholder={placeholder}
      onChange={e=>set(key,e.target.value)}
      style={{width:"100%",padding:"5px 8px",border:"1.5px solid #e2e8f0",borderRadius:7,
        fontSize:11.5,fontFamily:"'DM Sans',sans-serif",outline:"none",
        color:"#0f172a",background:"#fff",boxSizing:"border-box",textTransform:"uppercase"}}
      onFocus={e=>e.target.style.borderColor=PURPLE_HEX}
      onBlur={e=>e.target.style.borderColor="#e2e8f0"}
    />
  );

  return (
    <div style={{background:"#f8fafc",border:"1px solid #e2e8f0",borderRadius:10,overflow:"hidden"}}>
      {/* Toggle */}
      <button onClick={()=>setOpen(v=>!v)} style={{
        width:"100%",padding:"9px 12px",background:"none",border:"none",cursor:"pointer",
        display:"flex",alignItems:"center",justifyContent:"space-between",
        fontFamily:"'DM Sans',sans-serif",fontSize:12,fontWeight:600,color:"#334155",
      }}>
        <span style={{display:"flex",alignItems:"center",gap:7}}>
          <span style={{width:18,height:18,borderRadius:5,background:PURPLE_HEX,display:"inline-flex",alignItems:"center",justifyContent:"center",color:"#fff",fontSize:10,fontWeight:800,flexShrink:0}}>T</span>
          Personalizar cabeçalhos da tabela
        </span>
        <span style={{fontSize:10,opacity:.5}}>{open?"▼":"▶"}</span>
      </button>

      {open && (
        <div style={{padding:"0 12px 14px",borderTop:"1px solid #e2e8f0"}}>

          {/* ── TOP HEADER LINES ── */}
          <div style={{fontSize:10.5,fontWeight:700,color:"#334155",marginTop:12,marginBottom:8,display:"flex",alignItems:"center",justifyContent:"space-between"}}>
            <span>Linhas do cabeçalho (abaixo do logotipo)</span>
            <button onClick={addLine} style={{padding:"2px 10px",fontSize:11,fontWeight:600,borderRadius:7,cursor:"pointer",fontFamily:"'DM Sans',sans-serif",background:PURPLE_HEX,color:"#fff",border:"none"}}>
              + Adicionar linha
            </button>
          </div>
          <div style={{display:"flex",flexDirection:"column",gap:5,marginBottom:12}}>
            {headerLines.map((line,idx)=>(
              <div key={line.id}
                draggable
                onDragStart={()=>setDragIdx(idx)}
                onDragOver={e=>{e.preventDefault();}}
                onDrop={e=>{e.preventDefault();if(dragIdx!==null&&dragIdx!==idx){moveLine(dragIdx,idx);setDragIdx(null);}}}
                onDragEnd={()=>setDragIdx(null)}
                style={{display:"flex",alignItems:"center",gap:6,padding:"5px 8px",
                  background:"#fff",borderRadius:8,border:"1.5px solid #e2e8f0",
                  opacity:dragIdx===idx?0.4:1,cursor:"grab",
                  boxShadow:"0 1px 3px rgba(0,0,0,.06)"}}>
                {/* Drag handle */}
                <span style={{color:"#cbd5e1",fontSize:13,cursor:"grab",flexShrink:0,userSelect:"none"}}>⠿</span>
                {/* Text input — disabled if isSchool */}
                {line.isSchool
                  ?<div style={{flex:1,padding:"4px 8px",background:"#f1f5f9",borderRadius:6,fontSize:11.5,color:"#64748b",fontFamily:"'DM Sans',sans-serif",border:"1px solid #e2e8f0"}}>
                    Nome da escola <span style={{fontSize:10,color:"#94a3b8"}}>(automático)</span>
                  </div>
                  :<input type="text" value={line.text} onChange={e=>updateLine(line.id,"text",e.target.value)}
                    placeholder="Texto da linha…"
                    style={{flex:1,padding:"4px 8px",border:"1.5px solid #e2e8f0",borderRadius:6,fontSize:11.5,fontFamily:"'DM Sans',sans-serif",outline:"none",color:"#0f172a",background:"#fff"}}
                    onFocus={e=>e.target.style.borderColor=PURPLE_HEX}
                    onBlur={e=>e.target.style.borderColor="#e2e8f0"}
                  />
                }
                {/* Bold toggle */}
                <button onClick={()=>updateLine(line.id,"bold",!line.bold)}
                  title="Negrito"
                  style={{padding:"3px 8px",borderRadius:6,border:`1.5px solid ${line.bold?PURPLE_HEX:"#e2e8f0"}`,
                    background:line.bold?`rgba(96,81,255,.1)`:"#f8fafc",
                    color:line.bold?PURPLE_HEX:"#94a3b8",fontWeight:800,fontSize:12,cursor:"pointer",flexShrink:0,fontFamily:"serif"}}>

function ClassSlot({ slotId, allClasses, onRemove, onDataChange, onSubjSelChange }) {
  const [selectedClass,    setSelectedClass]    = useState("");
  const [studentsOptions,  setStudentsOptions]  = useState([]);
  const [selectedStudents, setSelectedStudents] = useState([]);
  const [loading,          setLoading]          = useState(false);
  const [data,             setData]            = useState(null);
  const [selectedSubjects, setSelectedSubjects] = useState([]);

  const handleClassChange = (e) => {
    const val=e.target.value;
    setSelectedClass(val); setData(null); setSelectedSubjects([]);
    onDataChange(slotId,null);
    if(!val){setStudentsOptions([]);setSelectedStudents([]);return;}
    axios.get(`${Hoot()}eduallgetstudentsbyclass/${val}`)
      .then(r=>{const list=(r.data||[]).map(s=>({value:String(s.ed_student_id),label:s.ed_student_name}));setStudentsOptions(list);setSelectedStudents(list);})
      .catch(()=>{setStudentsOptions([]);setSelectedStudents([]);});
  };

  useEffect(()=>{
    if(!selectedClass||!selectedStudents?.length){setData(null);onDataChange(slotId,null);return;}
    let cancelled=false; setLoading(true);
    axios.post(API.classGrades(),{classId:Number(selectedClass),studentIds:selectedStudents.map(s=>Number(s.value))})
      .then(res=>{
        if(cancelled)return;
        if(res.data?.success){
          const d={...res.data,_slotId:slotId};
          setData(d);
          const subs=res.data.data?.students?.[0]?.grades?.map(g=>g.subject_id)||[];
          setSelectedSubjects(subs);
          onDataChange(slotId,d);
          onSubjSelChange(slotId,subs);
        } else {toast.error(res.data?.message||"Erro.");setData(null);onDataChange(slotId,null);}
      })
      .catch(()=>{if(!cancelled){toast.error("Erro de comunicação.");setData(null);onDataChange(slotId,null);}})
      .finally(()=>{if(!cancelled)setLoading(false);});
    return()=>{cancelled=true;};
  },[selectedClass,selectedStudents]);

  const allSubjects=data?.data?.students?.[0]?.grades?.map(g=>({id:g.subject_id,title:g.subject_title,is_key:g.is_key}))||[];
  const toggleSubj=(id)=>{const next=selectedSubjects.includes(id)?selectedSubjects.filter(x=>x!==id):[...selectedSubjects,id];setSelectedSubjects(next);onSubjSelChange(slotId,next);};
  const toggleAll=()=>{const next=selectedSubjects.length===allSubjects.length?[]:allSubjects.map(s=>s.id);setSelectedSubjects(next);onSubjSelChange(slotId,next);};
  const clsLabel=allClasses.find(c=>c.value==selectedClass)?.label||"";

  return(
    <div className="cgr-class-card">
      <div className="cgr-class-card-hdr">
        <div style={{display:"flex",alignItems:"center",gap:9}}>
          <div style={{width:22,height:22,borderRadius:6,background:PURPLE_HEX,display:"flex",alignItems:"center",justifyContent:"center",color:"#fff",fontSize:11,fontWeight:700}}>{slotId}</div>
          <span style={{fontSize:13,fontWeight:600,color:"#334155"}}>{clsLabel||"Nova turma"}</span>
          {loading&&<Spinner size="sm" animation="border" style={{width:13,height:13,borderWidth:2}}/>}
        </div>
        <button className="cgr-rm-btn" onClick={()=>onRemove(slotId)} title="Remover"><MdOutlineClose size={15}/></button>
      </div>
      <div className="cgr-class-card-body">
        <div className="cgr-row">
          <div className="cgr-col">
            <div className="cgr-lbl"><MdOutlineClass size={12}/> Turma</div>
            <select className="cgr-sel" value={selectedClass} onChange={handleClassChange}>
              <option value="">Selecione…</option>
              {allClasses.map(c=><option key={c.value} value={c.value}>{c.label}</option>)}
            </select>
          </div>
          <div className="cgr-col" style={{flex:2}}>
            <div className="cgr-lbl"><MdOutlinePersonSearch size={12}/> Alunos</div>
            {selectedClass
              ?(studentsOptions.length>0?<StudentMultiSelect options={studentsOptions} value={selectedStudents} onChange={setSelectedStudents}/>:<div style={{fontSize:12,color:"#94a3b8"}}>A carregar…</div>)
              :<div style={{fontSize:12,color:"#94a3b8"}}>Seleccione uma turma.</div>}
          </div>
        </div>
        {allSubjects.length>0&&(
          <div style={{marginTop:12}}>
            <div style={{display:"flex",alignItems:"center",justifyContent:"space-between",marginBottom:7}}>
              <div className="cgr-lbl" style={{marginBottom:0}}><MdOutlineFilterList size={12}/> Disciplinas</div>
              <button onClick={toggleAll} style={{background:PURPLE_HEX,color:"#fff",border:"none",cursor:"pointer",fontSize:11,fontFamily:F_SANS,fontWeight:600,padding:"4px 12px",borderRadius:999,boxShadow:"0 2px 6px rgba(96,81,255,.32)",transition:"filter .15s"}}
                onMouseEnter={e=>e.currentTarget.style.filter="brightness(1.1)"} onMouseLeave={e=>e.currentTarget.style.filter="none"}>
                {selectedSubjects.length===allSubjects.length?"Remover todas":"Seleccionar todas"}
              </button>
            </div>
            <div className="cgr-subj-grid">
              {allSubjects.map(s=>(
                <button key={s.id} className={`cgr-subj-chip${selectedSubjects.includes(s.id)?" sel":""}`} onClick={()=>toggleSubj(s.id)}>
                  {selectedSubjects.includes(s.id)?<MdOutlineCheckCircle size={13}/>:<MdOutlineRadioButtonUnchecked size={13}/>}
                  <span style={{flex:1,fontSize:11.5,overflow:"hidden",whiteSpace:"nowrap",textOverflow:"ellipsis"}}>{s.title?.length>18?s.title.slice(0,18)+"…":s.title}</span>
                  {s.is_key&&<span className="cgr-key-badge">★</span>}
                </button>
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
